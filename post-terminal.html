2. ⭐ Starship 提示符

Starship 是用 Rust 写的跨 shell 提示符工具，速度快，配置简单：

# 安装
curl -sS https://starship.rs/install.sh | sh

# 加到 shell（以 zsh 为例）
echo 'eval "$(starship init zsh)"' >> ~/.zshrc

配置文件 ~/.config/starship.toml：

# ~/.config/starship.toml

format = """
[](fg:#89b4fa)
[  $username](bg:#89b4fa fg:#1e1e2e)
[](bg:#89b4fa fg:#a6e3a1)
[ $directory](bg:#a6e3a1 fg:#1e1e2e)
[](bg:#a6e3a1 fg:#f9e2af)
[ $git_branch $git_status](bg:#f9e2af fg:#1e1e2e)
[](bg:#f9e2af fg:#45475a)
[ $time](bg:#45475a fg:#cdd6f4)
[]
$character"""

[username]
show_always = true
style_user = "bold"
format = "[$user]($style)"

[directory]
style = "bold"
format = "[$path]($style) "
truncation_length = 3

[git_branch]
symbol = ""
format = "[$symbol $branch]($style) "

[git_status]
format = "([$all_status$ahead_behind]($style)) "

[time]
disabled = false
time_format = "%R"
format = "[$time]($style) "

[character]
success_symbol = "[╰─➤](bold green)"
error_symbol = "[╰─➤](bold red)"

3. 🐚 Zsh + 插件

# 安装 zsh
sudo apt install zsh  # Debian/Ubuntu
sudo pacman -S zsh    # Arch

# 安装 Oh My Zsh
sh -c "$(curl -fsSL https://raw.github.com/ohmyzsh/ohmyzsh/master/tools/install.sh)"

# 必装插件
git clone https://github.com/zsh-users/zsh-autosuggestions \
  ${ZSH_CUSTOM:-~/.oh-my-zsh/custom}/plugins/zsh-autosuggestions

git clone https://github.com/zsh-users/zsh-syntax-highlighting \
  ${ZSH_CUSTOM:-~/.oh-my-zsh/custom}/plugins/zsh-syntax-highlighting

启用插件（~/.zshrc）：

plugins=(git zsh-autosuggestions zsh-syntax-highlighting)

4. 📟 Tmux 会话管理

Tmux 让你在断开 SSH 后保持终端状态，还能分屏。配置 ~/.tmux.conf：

# ~/.tmux.conf

# 前缀键改为 Ctrl+a（比 Ctrl+b 顺手）
unbind C-b
set -g prefix C-a
bind C-a send-prefix

# 256 色支持
set -g default-terminal "tmux-256color"
set -ag terminal-overrides ",xterm-256color:RGB"

# 分屏快捷键
bind | split-window -h -c "#{pane_current_path}"
bind - split-window -v -c "#{pane_current_path}"

# vim 风格切换面板
bind h select-pane -L
bind j select-pane -D
bind k select-pane -U
bind l select-pane -R

# 鼠标支持
set -g mouse on

# 状态栏
set -g status-style "bg=#1e1e2e,fg=#cdd6f4"
set -g status-left " #S "
set -g status-left-style "bg=#89b4fa,fg=#1e1e2e,bold"
set -g status-right " %H:%M "
set -g status-right-style "bg=#45475a,fg=#cdd6f4"
set -g window-status-current-style "bg=#a6e3a1,fg=#1e1e2e,bold"
set -g window-status-format " #I: #W "
set -g window-status-current-format " #I: #W "

# 重载配置
bind r source-file ~/.tmux.conf \; display "Config reloaded!"

6. 🔤 Nerd Font 安装

Nerd Font 在普通编程字体基础上加入了图标支持，Starship 的图标全靠它：

# 下载 JetBrains Mono Nerd Font
mkdir -p ~/.local/share/fonts
cd ~/.local/share/fonts
curl -fLo "JetBrainsMono.zip" \
  https://github.com/ryanoasis/nerd-fonts/releases/latest/download/JetBrainsMono.zip
unzip JetBrainsMono.zip
rm JetBrainsMono.zip

# 刷新字体缓存
fc-cache -fv

🚀 总结

终端美化的核心是一致性：所有工具用同一套配色、同一套字体、同一种风格。不需要每个都换，先从 Starship 开始，效果立竿见影。
